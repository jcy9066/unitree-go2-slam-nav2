cmake_minimum_required(VERSION 3.8)
project(go2_nav2_critics)
find_package(ament_cmake REQUIRED)
find_package(dwb_core REQUIRED)
find_package(dwb_msgs REQUIRED)
find_package(nav2_util REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(${PROJECT_NAME} SHARED src/walking_speed.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
ament_target_dependencies(${PROJECT_NAME} dwb_core dwb_msgs nav2_util pluginlib rclcpp)
pluginlib_export_plugin_description_file(dwb_core critics.xml)
install(TARGETS ${PROJECT_NAME} ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
ament_export_libraries(${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(dwb_critics REQUIRED)
  find_package(nav2_costmap_2d REQUIRED)
  ament_add_gtest(test_motion_guards test/test_motion_guards.cpp)
  ament_target_dependencies(test_motion_guards dwb_core dwb_msgs dwb_critics nav2_costmap_2d pluginlib)
endif()
ament_package()
