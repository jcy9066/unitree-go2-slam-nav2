공식문서

#!/usr/bin/env python3
"""SLAM이 계산한 위치(ROS 좌표)를 PX4 좌표로 바꿔 EKF2에 전달한다.

ROS: x 앞, y 왼쪽, z 위 (FLU)   /   PX4: x 앞, y 오른쪽, z 아래 (FRD)
SLAM은 '출발할 때의 기체 자세'를 원점으로 잡는다. 그래서
 - 출발할 때 기체가 수평이어야 한다(평평한 바닥에서 켤 것).
 - PX4에는 POSE_FRAME_FRD(출발 방향 기준, 북쪽 모름)로 보낸다.
"""
import math

import rclpy
from nav_msgs.msg import Odometry
from px4_msgs.msg import VehicleOdometry
from rclpy.clock import Clock, ClockType
from rclpy.node import Node
from rclpy.qos import QoSHistoryPolicy, QoSProfile, QoSReliabilityPolicy
from rclpy.time import Time

LOST = 9999.0  # RTAB-Map은 추적을 잃거나 값이 없으면 공분산을 9999 이상으로 채운다
NAN3 = [math.nan, math.nan, math.nan]


def vec_flu_to_frd(x, y, z):
    """벡터: y와 z의 부호만 뒤집으면 된다."""
    return [x, -y, -z]


def quat_flu_to_frd(w, x, y, z):
    """자세: x축 180도 회전으로 앞뒤에서 감싼 결과는 y, z 부호 반전과 같다."""
    return [w, x, -y, -z]


def diag(cov, idx):
    return [float(cov[i]) for i in idx]


class OdomToPx4(Node):
    def __init__(self):
        super().__init__("odom_to_px4")
        self.declare_parameter("input_topic", "/odom")
        self.declare_parameter("max_latency_s", 0.3)
        topic = self.get_parameter("input_topic").value
        self.max_latency = float(self.get_parameter("max_latency_s").value)
        # PX4와 시간을 맞추는 통역기(Agent)는 PC의 실제 시계를 쓴다 -> 보내는 시각도 실제 시계
        self.wall = Clock(clock_type=ClockType.SYSTEM_TIME)

        qos = QoSProfile(
            reliability=QoSReliabilityPolicy.BEST_EFFORT,
            history=QoSHistoryPolicy.KEEP_LAST,
            depth=1,
        )
        self.pub = self.create_publisher(VehicleOdometry, "/fmu/in/vehicle_visual_odometry", qos)
        self.sub = self.create_subscription(Odometry, topic, self.on_odom, 10)
        self.sent = self.lost = self.late = 0
        self.latency_sum = 0.0
        self.create_timer(5.0, self.report)
        self.get_logger().info(f"입력 {topic} -> /fmu/in/vehicle_visual_odometry")

    def on_odom(self, msg):
        pcov = diag(msg.pose.covariance, (0, 7, 14))
        qcov = diag(msg.pose.covariance, (21, 28, 35))
        vcov = diag(msg.twist.covariance, (0, 7, 14))
        p, q = msg.pose.pose.position, msg.pose.pose.orientation
        pose_vals = pcov + qcov + [p.x, p.y, p.z, q.w, q.x, q.y, q.z]
        # NaN/무한대는 비교(>=)를 그냥 통과하므로 따로 막는다
        if not all(map(math.isfinite, pose_vals)) or max(pcov + qcov) >= LOST:
            self.lost += 1  # 틀린 위치를 넣느니 안 넣는다
            return

        # 영상이 찍힌 뒤 지금까지 걸린 시간(노드 시계 기준: 시뮬이면 시뮬 시계)
        latency = (self.get_clock().now() - Time.from_msg(msg.header.stamp)).nanoseconds * 1e-9
        if latency < -0.05 or latency > self.max_latency:
            self.late += 1
            return

        now_us = self.wall.now().nanoseconds // 1000
        out = VehicleOdometry()
        out.timestamp = now_us
        out.timestamp_sample = now_us - int(max(latency, 0.0) * 1e6)
        out.pose_frame = VehicleOdometry.POSE_FRAME_FRD
        out.velocity_frame = VehicleOdometry.VELOCITY_FRAME_BODY_FRD

        v, r = msg.twist.twist.linear, msg.twist.twist.angular
        out.position = [float(a) for a in vec_flu_to_frd(p.x, p.y, p.z)]
        out.q = [float(a) for a in quat_flu_to_frd(q.w, q.x, q.y, q.z)]
        out.position_variance = pcov  # 부호를 뒤집어도 분산(제곱)은 그대로
        out.orientation_variance = qcov
        vel_vals = vcov + [v.x, v.y, v.z, r.x, r.y, r.z]
        if not all(map(math.isfinite, vel_vals)) or max(vcov) >= LOST:  # 속도를 못 쟀으면 0이 아니라 '모름(NaN)'으로
            out.velocity = NAN3
            out.velocity_variance = NAN3
            out.angular_velocity = NAN3
        else:
            out.velocity = [float(a) for a in vec_flu_to_frd(v.x, v.y, v.z)]
            out.velocity_variance = vcov
            out.angular_velocity = [float(a) for a in vec_flu_to_frd(r.x, r.y, r.z)]
        self.pub.publish(out)
        self.sent += 1
        self.latency_sum += latency

    def report(self):
        avg = (self.latency_sum / self.sent * 1000) if self.sent else 0.0
        self.get_logger().info(
            f"최근 5초: 전달 {self.sent / 5.0:.1f} Hz, 평균 지연 {avg:.0f} ms, "
            f"추적 실패로 버림 {self.lost}, 너무 늦어 버림 {self.late}"
        )
        if self.sent == 0:
            self.get_logger().error("PX4로 보낸 위치가 없습니다. 이 상태로 이륙하면 안 됩니다.")
        self.sent = self.lost = self.late = 0
        self.latency_sum = 0.0


def main():
    rclpy.init()
    node = OdomToPx4()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()

        """일시정지 신호가 켜져 있거나, 감시 중인데 신호가 끊겼으면 True(안전한 쪽으로)."""
        stale = self.pause_timeout > 0 and self.age(self.pause_t) > self.pause_timeout
        if stale != self.pause_stale:
            if stale:
                self.get_logger().warning(f"일시정지 신호가 {self.pause_timeout:.1f}초 넘게 끊김(YOLO 꺼짐?): 제자리 대기")
            else:
                self.get_logger().info("일시정지 신호 다시 수신")
            self.pause_stale = stale
        return self.paused or stale

    # ---------- 송신 ----------
    def command(self, cmd, p1=0.0, p2=0.0):
        m = VehicleCommand()
        m.command = cmd
        m.param1, m.param2 = float(p1), float(p2)
        m.target_system = m.target_component = 1
        m.source_system = m.source_component = 1
        m.from_external = True
        m.timestamp = self.now().nanoseconds // 1000
        self.cmd_pub.publish(m)

    def send_setpoint(self, xyz):
        t = self.now().nanoseconds // 1000
        mode = OffboardControlMode()
        mode.position = True
        mode.timestamp = t
        self.mode_pub.publish(mode)
        sp = TrajectorySetpoint()
        nan = float("nan")
        sp.position = [float(v) for v in xyz]
        sp.velocity = [nan, nan, nan]
        sp.acceleration = [nan, nan, nan]
        sp.yaw = float(self.home_yaw)
        sp.timestamp = t
        self.sp_pub.publish(sp)

    # ---------- 판단 ----------
    def problems(self):
        p = []
        if self.age(self.pos_t) > FRESH_S or not (self.pos.xy_valid and self.pos.z_valid):
            p.append("PX4 위치 추정 없음")
        if self.age(self.status_t) > FRESH_S:
            p.append("PX4 상태 메시지 끊김")
        if self.require_vision:
            if self.age(self.flags_t) > FLAGS_FRESH_S:
                p.append("EKF2 상태 메시지 끊김")
            elif not self.vision_ok:
                p.append("EKF2가 비전 위치를 쓰지 않음(cs_ev_pos/cs_ev_hgt)")
        return p

    def step_toward(self, goal):
        """보내는 목표점을 경유점 쪽으로 0.1초 분량(speed x 0.1 m)만 옮긴다."""
        d = math.dist(self.cmd, goal)
        stride = self.speed * 0.1
        if d <= stride:
            self.cmd = list(goal)
        else:
            self.cmd = [c + (g - c) * stride / d for c, g in zip(self.cmd, goal)]

    def start_landing(self, why):
        self.get_logger().error(f"{why}. 착륙합니다.") if why else self.get_logger().info("착륙합니다.")
        self.phase = "LANDING"
        self.ticks = 0

    def start_disarm(self, why):
        self.get_logger().error(f"{why}. 시동을 끕니다.")
        self.phase = "DISARMING"
        self.ticks = 0

    def tick_landing(self, st):
        fresh = st is not None and self.age(self.status_t) < FRESH_S
        if not fresh:
            # 상태를 모르면 '착륙 중'이라고 믿지 않는다. 착륙 요청을 계속 보낸다.
            if self.ticks % 10 == 1:
                self.command(VehicleCommand.VEHICLE_CMD_NAV_LAND)
            if self.ticks % 50 == 1:
                self.get_logger().warning("PX4 상태 메시지가 끊겨 착륙 여부를 확인할 수 없습니다.")
            if self.ticks > 100:
                self.finish("10초 동안 PX4 상태를 확인하지 못했습니다. 즉시 조종기로 착륙하세요!", True)
        elif st.arming_state == VS.ARMING_STATE_DISARMED:
            self.finish("착륙 및 시동 해제 확인.")
        elif st.nav_state in (VS.NAVIGATION_STATE_AUTO_LAND, VS.NAVIGATION_STATE_DESCEND):
            if self.ticks % 50 == 1:
                kind = "착륙(Land)" if st.nav_state == VS.NAVIGATION_STATE_AUTO_LAND else "비상 하강(Descend, 위치 없이 내려옴)"
                self.get_logger().info(f"PX4가 {kind} 중입니다. 시동 해제를 기다립니다.")
            if self.ticks > 900:
                self.finish("90초가 지나도 시동이 꺼지지 않았습니다. 조종기로 직접 확인하세요!", True)
        elif st.nav_state == VS.NAVIGATION_STATE_OFFBOARD:
            if self.ticks % 10 == 1:  # 아직 내 명령 아래 있음 -> 1초마다 착륙 요청
                self.command(VehicleCommand.VEHICLE_CMD_NAV_LAND)
            if self.ticks > 100:
                self.finish("10초 동안 착륙 모드로 바뀌지 않았습니다. 즉시 조종기로 착륙하세요!", True)
        else:
            # 조종자(또는 PX4 안전장치)가 다른 모드로 바꿨다 -> 덮어쓰지 않는다
            self.finish("착륙 중 다른 모드로 바뀌었습니다(조종자 인계 또는 PX4 안전장치). 프로그램은 더 이상 명령하지 않습니다.", True)

    def finish(self, msg, error=False):
        (self.get_logger().error if error else self.get_logger().info)(msg)
        self.phase = "DONE"

    def tick(self):
        self.ticks += 1
        st = self.status
        if self.phase == "DONE":
            rclpy.shutdown()
            return

        if self.phase == "WAIT_READY":
            if st is not None and st.arming_state == VS.ARMING_STATE_ARMED:
                self.finish("이미 시동이 걸려 있습니다. 착륙·시동 해제 후 다시 실행하세요.", True)
                return
            probs = self.problems() if self.pos is not None else ["PX4 위치 추정 없음"]
            if self.is_paused():
                probs.append("일시정지 신호가 켜져 있거나 끊김")
            if not probs:
                x, y, z = self.pos.x, self.pos.y, self.pos.z
                self.home_yaw = self.pos.heading
                c, s = math.cos(self.home_yaw), math.sin(self.home_yaw)
                top = z - self.alt
                pts = [(0, 0), (self.side, 0), (self.side, self.side), (0, self.side), (0, 0)]
                self.waypoints = [[x + c * a - s * b, y + s * a + c * b, top] for a, b in pts]
                self.cmd = [x, y, z]  # 지금 서 있는 자리에서 출발
                self.phase, self.ticks = "ARMING", 0
                self.get_logger().info("준비 완료. Offboard 전환과 시동을 요청합니다.")
            elif self.ticks % 50 == 1:
                self.get_logger().warning("대기 중: " + ", ".join(probs))
            return

        if self.phase == "LANDING":
            self.tick_landing(st)
            return

        if self.phase == "DISARMING":
            fresh = st is not None and self.age(self.status_t) < FRESH_S
            if fresh and st.arming_state == VS.ARMING_STATE_DISARMED:
                self.finish("시동 해제 확인. 프로그램을 끝냅니다.", True)
            else:
                if self.ticks % 10 == 1:  # 1초마다 다시 요청
                    self.command(VehicleCommand.VEHICLE_CMD_COMPONENT_ARM_DISARM, 0)
                if self.ticks > 100:
                    self.finish("10초 동안 시동 해제가 확인되지 않았습니다. 즉시 조종기로 시동을 끄세요!", True)
            return

        probs = self.problems()
        paused = self.is_paused()
        if paused:
            if self.hold is None and self.pos is not None:
                self.hold = [self.pos.x, self.pos.y, self.pos.z]
            if self.hold is not None:
                self.cmd = list(self.hold)  # 멈춘 자리에서 다시 출발하도록
        else:
            self.hold = None
            if self.phase == "FLY":
                self.step_toward(self.waypoints[self.index])
        target = self.cmd

        if self.phase == "ARMING":
            if probs or paused:
                reason = ", ".join(probs) if probs else "일시정지 신호"
                if self.was_armed:
                    self.start_disarm(f"이륙 전 중단: {reason}")
                else:
                    self.finish(f"이륙 전 중단: {reason}", True)
                return
            self.send_setpoint(target)
            if self.ticks == 20:  # 2초간 목표점을 먼저 보내야 Offboard 전환이 허용된다
                self.command(VehicleCommand.VEHICLE_CMD_DO_SET_MODE, 1, 6)
                self.command(VehicleCommand.VEHICLE_CMD_COMPONENT_ARM_DISARM, 1)
            if st.arming_state == VS.ARMING_STATE_ARMED and st.nav_state == VS.NAVIGATION_STATE_OFFBOARD:
                self.get_logger().info("Offboard 모드 시동 성공. 이륙합니다.")
                self.phase = "FLY"
            elif self.ticks > 100:
                why = "10초 안에 Offboard 시동이 되지 않았습니다. QGC 경고를 확인하세요"
                if self.was_armed:
                    self.start_disarm(why)
                else:
                    self.finish(why + ".", True)
            return

        # FLY
        if st.arming_state != VS.ARMING_STATE_ARMED or st.nav_state != VS.NAVIGATION_STATE_OFFBOARD:
            self.finish("Offboard가 해제됐습니다(조종자 인계 또는 PX4 안전장치). 프로그램은 더 이상 명령하지 않습니다.", True)
            return
        if probs:
            self.start_landing("비행 중 이상: " + ", ".join(probs))
            return
        self.send_setpoint(target)
        goal = self.waypoints[self.index]
        if not paused and self.cmd == goal and math.dist([self.pos.x, self.pos.y, self.pos.z], goal) < 0.25:
            self.get_logger().info(f"경유점 {self.index + 1}/{len(self.waypoints)} 도착")
            if self.index + 1 >= len(self.waypoints):
                self.start_landing("")
            else:
                self.index += 1


def main():
    rclpy.init()

#!/usr/bin/env python3
"""YOLO로 카메라 영상 속 물체를 찾고, 사람이 가까이 있으면 비행을 일시정지시킨다.

입력 : image_topic (기본 /camera, 실기체 RealSense는 /camera/camera/color/image_raw)
출력 : /yolo/image (네모 박스가 그려진 영상), /yolo/detections (글자), /safety/pause (Bool)
"""
import time

import numpy as np
import rclpy
import torch
from rclpy.node import Node
from sensor_msgs.msg import Image
from std_msgs.msg import Bool, String
from ultralytics import YOLO


class YoloDetect(Node):
    def __init__(self):
        super().__init__("yolo_detect")
        self.declare_parameter("image_topic", "/camera")
        self.declare_parameter("model", "yolo11n.pt")
        self.declare_parameter("confidence", 0.5)
        self.declare_parameter("pause_class", "person")
        self.declare_parameter("pause_height_ratio", 0.4)  # 박스 높이가 화면의 40% 이상이면 '가깝다'
        self.declare_parameter("release_after_s", 1.5)
        self.conf = float(self.get_parameter("confidence").value)
        self.pause_class = self.get_parameter("pause_class").value
        self.ratio = float(self.get_parameter("pause_height_ratio").value)
        self.release_after = float(self.get_parameter("release_after_s").value)

        # GPU(CUDA)가 있으면 0번 GPU, 없으면 CPU. 조용히 바꾸지 않고 어느 쪽인지 로그로 알린다.
        self.device = 0 if torch.cuda.is_available() else "cpu"
        self.model = YOLO(self.get_parameter("model").value)
        where = torch.cuda.get_device_name(0) if self.device == 0 else "CPU (GPU를 못 찾음: 느림)"
        self.get_logger().info(f"모델 로드 완료, 계산 장치: {where}")
        self.img_pub = self.create_publisher(Image, "/yolo/image", 1)
        self.det_pub = self.create_publisher(String, "/yolo/detections", 10)
        self.pause_pub = self.create_publisher(Bool, "/safety/pause", 10)
        self.create_subscription(Image, self.get_parameter("image_topic").value, self.on_image, 1)
        self.last_close = 0.0

    def to_bgr(self, msg):
        if msg.encoding not in ("rgb8", "bgr8"):
            raise ValueError(f"지원하지 않는 영상 형식: {msg.encoding}")
        frame = np.frombuffer(msg.data, dtype=np.uint8).reshape(msg.height, msg.step)
        frame = frame[:, : msg.width * 3].reshape(msg.height, msg.width, 3)
        return frame[:, :, ::-1].copy() if msg.encoding == "rgb8" else frame.copy()

    def on_image(self, msg):
        try:
            bgr = self.to_bgr(msg)
        except ValueError as e:
            self.get_logger().error(str(e))
            return
        result = self.model(bgr, conf=self.conf, device=self.device, verbose=False)[0]
        names = result.names
        found = []
        close = False
        for box in result.boxes:
            label = names[int(box.cls)]
            x1, y1, x2, y2 = box.xyxy[0].tolist()
            found.append(f"{label}:{float(box.conf):.2f}")
            if label == self.pause_class and (y2 - y1) / msg.height >= self.ratio:
                close = True
        if close:
            self.last_close = time.monotonic()
        paused = (time.monotonic() - self.last_close) < self.release_after
        self.pause_pub.publish(Bool(data=paused))
        self.det_pub.publish(String(data=", ".join(found) if found else "(없음)"))

        drawn = result.plot()  # BGR
        out = Image()
        out.header = msg.header
        out.height, out.width = drawn.shape[:2]
        out.encoding = "bgr8"
        out.step = out.width * 3
        out.data = drawn.tobytes()
        self.img_pub.publish(out)


def main():
    rclpy.init()
    node = YoloDetect()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()

        self.declare_parameter("require_vision", True)
        self.declare_parameter("speed", 0.5)
        # >0 이면 /safety/pause(YOLO 등)가 이 시간(초) 넘게 안 오면 '정지'로 본다. 0이면 감시 안 함.
        self.declare_parameter("pause_timeout_s", 0.0)
        self.alt = float(self.get_parameter("altitude").value)
        self.side = float(self.get_parameter("side").value)
        self.speed = float(self.get_parameter("speed").value)
        self.pause_timeout = float(self.get_parameter("pause_timeout_s").value)
        self.require_vision = bool(self.get_parameter("require_vision").value)
        limits = {"altitude": (self.alt, 0.3, 10.0), "side": (self.side, 0.2, 20.0),
                  "speed": (self.speed, 0.05, 3.0), "pause_timeout_s": (self.pause_timeout, 0.0, 10.0)}
        for name, (v, lo, hi) in limits.items():
            if not (math.isfinite(v) and lo <= v <= hi):
                raise ValueError(f"{name}={v} 은(는) 허용 범위 {lo}~{hi} 밖입니다")

        qos = QoSProfile(
            reliability=QoSReliabilityPolicy.BEST_EFFORT,
            durability=QoSDurabilityPolicy.TRANSIENT_LOCAL,
            history=QoSHistoryPolicy.KEEP_LAST,
            depth=1,
        )
        self.mode_pub = self.create_publisher(OffboardControlMode, "/fmu/in/offboard_control_mode", qos)
        self.sp_pub = self.create_publisher(TrajectorySetpoint, "/fmu/in/trajectory_setpoint", qos)
        self.cmd_pub = self.create_publisher(VehicleCommand, "/fmu/in/vehicle_command", qos)
        self.create_subscription(VehicleLocalPosition, "/fmu/out/vehicle_local_position_v1", self.on_pos, qos)
        self.create_subscription(VehicleStatus, "/fmu/out/vehicle_status_v1", self.on_status, qos)
        self.create_subscription(EstimatorStatusFlags, "/fmu/out/estimator_status_flags", self.on_flags, qos)
        self.create_subscription(Bool, "/safety/pause", self.on_pause, 10)

        self.pos = self.pos_t = None
        self.status = self.status_t = None
        self.vision_ok = False
        self.flags_t = None
        self.paused = False
        self.pause_t = None
        self.pause_stale = False
        self.hold = None
        self.cmd = None  # 지금 PX4에 보내는 목표점(경유점을 향해 조금씩 움직인다)
        self.ticks = 0
        self.home_yaw = 0.0
        self.waypoints = []
        self.index = 0
        self.was_armed = False
        self.phase = "WAIT_READY"
        self.create_timer(0.1, self.tick)
        mode = "비전(V-SLAM) 필수" if self.require_vision else "비전 확인 안 함(GPS 시험용)"
        if self.pause_timeout > 0:
            mode += f", 일시정지 신호 감시({self.pause_timeout:.1f}초 끊기면 정지)"
        self.get_logger().info(f"준비 상태를 기다립니다. 모드: {mode}")

    # ---------- 수신 ----------
    def now(self):
        return self.get_clock().now()

    def age(self, t):
        return math.inf if t is None else (self.now() - t).nanoseconds * 1e-9

    def on_pos(self, msg):
        self.pos, self.pos_t = msg, self.now()

    def on_status(self, msg):
        self.status, self.status_t = msg, self.now()
        if msg.arming_state == VS.ARMING_STATE_ARMED:
            self.was_armed = True

    def on_flags(self, msg):
        self.vision_ok = bool(msg.cs_ev_pos and msg.cs_ev_hgt)
        self.flags_t = self.now()

    def on_pause(self, msg):
        if msg.data and not self.paused:
            self.get_logger().warning("일시정지 신호 수신: 제자리 대기")
        elif not msg.data and self.paused:
            self.get_logger().info("일시정지 해제")
        self.paused = msg.data
        self.pause_t = self.now()

    def is_paused(self):
        """일시정지 신호가 켜져 있거나, 감시 중인데 신호가 끊겼으면 True(안전한 쪽으로)."""
        stale = self.pause_timeout > 0 and self.age(self.pause_t) > self.pause_timeout
        if stale != self.pause_stale:
            if stale:
                self.get_logger().warning(f"일시정지 신호가 {self.pause_timeout:.1f}초 넘게 끊김(YOLO 꺼짐?): 제자리 대기")
            else:
                self.get_logger().info("일시정지 신호 다시 수신")
            self.pause_stale = stale
        return self.paused or stale

    # ---------- 송신 ----------
    def command(self, cmd, p1=0.0, p2=0.0):
        m = VehicleCommand()
        m.command = cmd
        m.param1, m.param2 = float(p1), float(p2)
        m.target_system = m.target_component = 1
        m.source_system = m.source_component = 1
        m.from_external = True
        m.timestamp = self.now().nanoseconds // 1000
        self.cmd_pub.publish(m)

    def send_setpoint(self, xyz):
        t = self.now().nanoseconds // 1000
        mode = OffboardControlMode()
        mode.position = True
        mode.timestamp = t
        self.mode_pub.publish(mode)
        sp = TrajectorySetpoint()
        nan = float("nan")
        sp.position = [float(v) for v in xyz]
        sp.velocity = [nan, nan, nan]
        sp.acceleration = [nan, nan, nan]
        sp.yaw = float(self.home_yaw)
        sp.timestamp = t
        self.sp_pub.publish(sp)

    # ---------- 판단 ----------
    def problems(self):
        p = []
        if self.age(self.pos_t) > FRESH_S or not (self.pos.xy_valid and self.pos.z_valid):
            p.append("PX4 위치 추정 없음")
        if self.age(self.status_t) > FRESH_S:
            p.append("PX4 상태 메시지 끊김")
        if self.require_vision:
            if self.age(self.flags_t) > FLAGS_FRESH_S:
                p.append("EKF2 상태 메시지 끊김")
            elif not self.vision_ok:
                p.append("EKF2가 비전 위치를 쓰지 않음(cs_ev_pos/cs_ev_hgt)")
        return p

    def step_toward(self, goal):
        """보내는 목표점을 경유점 쪽으로 0.1초 분량(speed x 0.1 m)만 옮긴다."""
        d = math.dist(self.cmd, goal)
        stride = self.speed * 0.1
        if d <= stride:
            self.cmd = list(goal)
        else:
            self.cmd = [c + (g - c) * stride / d for c, g in zip(self.cmd, goal)]

    def start_landing(self, why):
        self.get_logger().error(f"{why}. 착륙합니다.") if why else self.get_logger().info("착륙합니다.")
        self.phase = "LANDING"
        self.ticks = 0

    def start_disarm(self, why):
        self.get_logger().error(f"{why}. 시동을 끕니다.")
        self.phase = "DISARMING"
        self.ticks = 0

    def tick_landing(self, st):
        fresh = st is not None and self.age(self.status_t) < FRESH_S
        if not fresh:
            # 상태를 모르면 '착륙 중'이라고 믿지 않는다. 착륙 요청을 계속 보낸다.
            if self.ticks % 10 == 1:
                self.command(VehicleCommand.VEHICLE_CMD_NAV_LAND)
            if self.ticks % 50 == 1:
                self.get_logger().warning("PX4 상태 메시지가 끊겨 착륙 여부를 확인할 수 없습니다.")
            if self.ticks > 100:
                self.finish("10초 동안 PX4 상태를 확인하지 못했습니다. 즉시 조종기로 착륙하세요!", True)
        elif st.arming_state == VS.ARMING_STATE_DISARMED:
            self.finish("착륙 및 시동 해제 확인.")
        elif st.nav_state in (VS.NAVIGATION_STATE_AUTO_LAND, VS.NAVIGATION_STATE_DESCEND):
            if self.ticks % 50 == 1:
                kind = "착륙(Land)" if st.nav_state == VS.NAVIGATION_STATE_AUTO_LAND else "비상 하강(Descend, 위치 없이 내려옴)"
                self.get_logger().info(f"PX4가 {kind} 중입니다. 시동 해제를 기다립니다.")
            if self.ticks > 900:
                self.finish("90초가 지나도 시동이 꺼지지 않았습니다. 조종기로 직접 확인하세요!", True)
        elif st.nav_state == VS.NAVIGATION_STATE_OFFBOARD:
            if self.ticks % 10 == 1:  # 아직 내 명령 아래 있음 -> 1초마다 착륙 요청
                self.command(VehicleCommand.VEHICLE_CMD_NAV_LAND)
            if self.ticks > 100:
                self.finish("10초 동안 착륙 모드로 바뀌지 않았습니다. 즉시 조종기로 착륙하세요!", True)
        else:
            # 조종자(또는 PX4 안전장치)가 다른 모드로 바꿨다 -> 덮어쓰지 않는다
            self.finish("착륙 중 다른 모드로 바뀌었습니다(조종자 인계 또는 PX4 안전장치). 프로그램은 더 이상 명령하지 않습니다.", True)

    def finish(self, msg, error=False):
        (self.get_logger().error if error else self.get_logger().info)(msg)
        self.phase = "DONE"

    def tick(self):
        self.ticks += 1
        st = self.status
        if self.phase == "DONE":
            rclpy.shutdown()
            return

        if self.phase == "WAIT_READY":
            if st is not None and st.arming_state == VS.ARMING_STATE_ARMED:
                self.finish("이미 시동이 걸려 있습니다. 착륙·시동 해제 후 다시 실행하세요.", True)
                return
            probs = self.problems() if self.pos is not None else ["PX4 위치 추정 없음"]
            if self.is_paused():
                probs.append("일시정지 신호가 켜져 있거나 끊김")
            if not probs:
                x, y, z = self.pos.x, self.pos.y, self.pos.z
                self.home_yaw = self.pos.heading
                c, s = math.cos(self.home_yaw), math.sin(self.home_yaw)
                top = z - self.alt
                pts = [(0, 0), (self.side, 0), (self.side, self.side), (0, self.side), (0, 0)]
                self.waypoints = [[x + c * a - s * b, y + s * a + c * b, top] for a, b in pts]
                self.cmd = [x, y, z]  # 지금 서 있는 자리에서 출발
                self.phase, self.ticks = "ARMING", 0
                self.get_logger().info("준비 완료. Offboard 전환과 시동을 요청합니다.")
            elif self.ticks % 50 == 1:
                self.get_logger().warning("대기 중: " + ", ".join(probs))
            return

        if self.phase == "LANDING":
            self.tick_landing(st)
            return

        if self.phase == "DISARMING":
            fresh = st is not None and self.age(self.status_t) < FRESH_S
            if fresh and st.arming_state == VS.ARMING_STATE_DISARMED:
                self.finish("시동 해제 확인. 프로그램을 끝냅니다.", True)
            else:
                if self.ticks % 10 == 1:  # 1초마다 다시 요청
                    self.command(VehicleCommand.VEHICLE_CMD_COMPONENT_ARM_DISARM, 0)
                if self.ticks > 100:
                    self.finish("10초 동안 시동 해제가 확인되지 않았습니다. 즉시 조종기로 시동을 끄세요!", True)
            return

        probs = self.problems()
        paused = self.is_paused()
        if paused:
            if self.hold is None and self.pos is not None:
                self.hold = [self.pos.x, self.pos.y, self.pos.z]
            if self.hold is not None:
                self.cmd = list(self.hold)  # 멈춘 자리에서 다시 출발하도록
        else:
            self.hold = None
            if self.phase == "FLY":
                self.step_toward(self.waypoints[self.index])
        target = self.cmd

        if self.phase == "ARMING":
            if probs or paused:
                reason = ", ".join(probs) if probs else "일시정지 신호"
                if self.was_armed:
                    self.start_disarm(f"이륙 전 중단: {reason}")
                else:
                    self.finish(f"이륙 전 중단: {reason}", True)
                return
            self.send_setpoint(target)
            if self.ticks == 20:  # 2초간 목표점을 먼저 보내야 Offboard 전환이 허용된다
                self.command(VehicleCommand.VEHICLE_CMD_DO_SET_MODE, 1, 6)
                self.command(VehicleCommand.VEHICLE_CMD_COMPONENT_ARM_DISARM, 1)
            if st.arming_state == VS.ARMING_STATE_ARMED and st.nav_state == VS.NAVIGATION_STATE_OFFBOARD:
                self.get_logger().info("Offboard 모드 시동 성공. 이륙합니다.")
                self.phase = "FLY"
            elif self.ticks > 100:
                why = "10초 안에 Offboard 시동이 되지 않았습니다. QGC 경고를 확인하세요"
                if self.was_armed:
                    self.start_disarm(why)
                else:
                    self.finish(why + ".", True)
            return

        # FLY
        if st.arming_state != VS.ARMING_STATE_ARMED or st.nav_state != VS.NAVIGATION_STATE_OFFBOARD:
            self.finish("Offboard가 해제됐습니다(조종자 인계 또는 PX4 안전장치). 프로그램은 더 이상 명령하지 않습니다.", True)
            return
        if probs:
            self.start_landing("비행 중 이상: " + ", ".join(probs))
            return
        self.send_setpoint(target)
        goal = self.waypoints[self.index]
        if not paused and self.cmd == goal and math.dist([self.pos.x, self.pos.y, self.pos.z], goal) < 0.25:
            self.get_logger().info(f"경유점 {self.index + 1}/{len(self.waypoints)} 도착")
            if self.index + 1 >= len(self.waypoints):
                self.start_landing("")
            else:
                self.index += 1

Offboard 비행의 규칙 세 가지 — 지키지 않으면 PX4가 명령을 무시한다

공식문서
docs.px4.io ros2/offboard_control · fly_square.py
  • 심장 박동: OffboardControlMode를 초당 2번 이상 계속 보내야 합니다. 끊기면 PX4가 Offboard를 빠져나갑니다(이 문서 설정: 1초 끊기면 착륙).
  • 먼저 보내고 나중에 전환: 목표점을 미리 보내고 있어야 Offboard 전환과 시동이 허용됩니다. 코드는 2초간 보낸 뒤 전환합니다.
  • 좌표는 NED: PX4는 x=북(앞), y=동(오른쪽), z=아래가 +입니다. 그래서 1.5m 위는 z = -1.5입니다. ROS(위가 +)와 반대이며 자동 변환은 없습니다.

    #!/usr/bin/env python3
"""SLAM이 계산한 위치(ROS 좌표)를 PX4 좌표로 바꿔 EKF2에 전달한다.

ROS: x 앞, y 왼쪽, z 위 (FLU)   /   PX4: x 앞, y 오른쪽, z 아래 (FRD)
SLAM은 '출발할 때의 기체 자세'를 원점으로 잡는다. 그래서
 - 출발할 때 기체가 수평이어야 한다(평평한 바닥에서 켤 것).
 - PX4에는 POSE_FRAME_FRD(출발 방향 기준, 북쪽 모름)로 보낸다.
"""
import math

import rclpy
from nav_msgs.msg import Odometry
from px4_msgs.msg import VehicleOdometry
from rclpy.clock import Clock, ClockType
from rclpy.node import Node
from rclpy.qos import QoSHistoryPolicy, QoSProfile, QoSReliabilityPolicy
from rclpy.time import Time

LOST = 9999.0  # RTAB-Map은 추적을 잃거나 값이 없으면 공분산을 9999 이상으로 채운다
NAN3 = [math.nan, math.nan, math.nan]


def vec_flu_to_frd(x, y, z):
    """벡터: y와 z의 부호만 뒤집으면 된다."""
    return [x, -y, -z]


def quat_flu_to_frd(w, x, y, z):
    """자세: x축 180도 회전으로 앞뒤에서 감싼 결과는 y, z 부호 반전과 같다."""
    return [w, x, -y, -z]


def diag(cov, idx):
    return [float(cov[i]) for i in idx]


class OdomToPx4(Node):
    def __init__(self):
        super().__init__("odom_to_px4")
        self.declare_parameter("input_topic", "/odom")
        self.declare_parameter("max_latency_s", 0.3)
        topic = self.get_parameter("input_topic").value
        self.max_latency = float(self.get_parameter("max_latency_s").value)
        # PX4와 시간을 맞추는 통역기(Agent)는 PC의 실제 시계를 쓴다 -> 보내는 시각도 실제 시계
        self.wall = Clock(clock_type=ClockType.SYSTEM_TIME)

        qos = QoSProfile(
            reliability=QoSReliabilityPolicy.BEST_EFFORT,
            history=QoSHistoryPolicy.KEEP_LAST,
            depth=1,
        )
        self.pub = self.create_publisher(VehicleOdometry, "/fmu/in/vehicle_visual_odometry", qos)
        self.sub = self.create_subscription(Odometry, topic, self.on_odom, 10)
        self.sent = self.lost = self.late = 0
        self.latency_sum = 0.0
        self.create_timer(5.0, self.report)
        self.get_logger().info(f"입력 {topic} -> /fmu/in/vehicle_visual_odometry")

    def on_odom(self, msg):
        pcov = diag(msg.pose.covariance, (0, 7, 14))
        qcov = diag(msg.pose.covariance, (21, 28, 35))
        vcov = diag(msg.twist.covariance, (0, 7, 14))
        p, q = msg.pose.pose.position, msg.pose.pose.orientation
        pose_vals = pcov + qcov + [p.x, p.y, p.z, q.w, q.x, q.y, q.z]
        # NaN/무한대는 비교(>=)를 그냥 통과하므로 따로 막는다
        if not all(map(math.isfinite, pose_vals)) or max(pcov + qcov) >= LOST:
            self.lost += 1  # 틀린 위치를 넣느니 안 넣는다
            return

        # 영상이 찍힌 뒤 지금까지 걸린 시간(노드 시계 기준: 시뮬이면 시뮬 시계)
        latency = (self.get_clock().now() - Time.from_msg(msg.header.stamp)).nanoseconds * 1e-9
        if latency < -0.05 or latency > self.max_latency:
            self.late += 1
            return

        now_us = self.wall.now().nanoseconds // 1000
        out = VehicleOdometry()
        out.timestamp = now_us
        out.timestamp_sample = now_us - int(max(latency, 0.0) * 1e6)
        out.pose_frame = VehicleOdometry.POSE_FRAME_FRD
        out.velocity_frame = VehicleOdometry.VELOCITY_FRAME_BODY_FRD

        v, r = msg.twist.twist.linear, msg.twist.twist.angular
        out.position = [float(a) for a in vec_flu_to_frd(p.x, p.y, p.z)]
        out.q = [float(a) for a in quat_flu_to_frd(q.w, q.x, q.y, q.z)]
        out.position_variance = pcov  # 부호를 뒤집어도 분산(제곱)은 그대로
        out.orientation_variance = qcov
        vel_vals = vcov + [v.x, v.y, v.z, r.x, r.y, r.z]
        if not all(map(math.isfinite, vel_vals)) or max(vcov) >= LOST:  # 속도를 못 쟀으면 0이 아니라 '모름(NaN)'으로
            out.velocity = NAN3
            out.velocity_variance = NAN3
            out.angular_velocity = NAN3
        else:
            out.velocity = [float(a) for a in vec_flu_to_frd(v.x, v.y, v.z)]
            out.velocity_variance = vcov
            out.angular_velocity = [float(a) for a in vec_flu_to_frd(r.x, r.y, r.z)]
        self.pub.publish(out)
        self.sent += 1
        self.latency_sum += latency

    def report(self):
        avg = (self.latency_sum / self.sent * 1000) if self.sent else 0.0
        self.get_logger().info(
            f"최근 5초: 전달 {self.sent / 5.0:.1f} Hz, 평균 지연 {avg:.0f} ms, "
            f"추적 실패로 버림 {self.lost}, 너무 늦어 버림 {self.late}"
        )
        if self.sent == 0:
            self.get_logger().error("PX4로 보낸 위치가 없습니다. 이 상태로 이륙하면 안 됩니다.")
        self.sent = self.lost = self.late = 0
        self.latency_sum = 0.0


def main():
    rclpy.init()
    node = OdomToPx4()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()

5-B

좌표 변환은 "y와 z 부호 뒤집기" 하나로 끝난다

검증됨
odom_to_px4.py · 무작위 자세 1000개 단위시험 최대오차 0, 왼쪽 90° 회전 → PX4 −90° 확인

ROS는 x 앞·y 왼쪽·z 위(FLU), PX4는 x 앞·y 오른쪽·z 아래(FRD)입니다. SLAM은 출발 자세를 원점으로 잡으므로, PX4에 POSE_FRAME_FRD(출발 방향 기준 좌표)로 보내면 북쪽을 몰라도 됩니다. 그래서 위치·속도·회전 모두 y, z 부호만 바꾸면 됩니다.

\[ \mathbf{p}_{\text{FRD}} = (x,\,-y,\,-z), \qquad q_{\text{FRD}} = (w,\,x,\,-y,\,-z) \]
SLAM이 길을 잃으면(RTAB-Map 공분산 9999) 그 값은 버리고 경고합니다. 틀린 위치를 넣는 것보다 안 넣는 것이 안전합니다. 0.3초보다 늦게 도착한 위치도 버립니다. fly_square.py는 PX4가 내보내는 "EKF2가 지금 카메라 위치를 쓰고 있나"(cs_ev_pos, cs_ev_hgt)를 계속 보다가, 꺼졌다는 소식이 오면 바로 착륙시킵니다. 이 소식은 평소 1초에 한 번만 오고 바뀔 때는 즉시 오므로(PX4 EKF2.cpp), 3초 넘게 소식이 없을 때도 끊김으로 봅니다. 목표점은 초당 0.5m씩만 옮깁니다(speed). 2m를 한 번에 던졌더니 기체가 급히 움직여 SLAM이 길을 잃는 것을 검증 중에 직접 봤습니다.
전제: 출발할 때 수평. 이 변환은 SLAM의 z축이 하늘을 향한다고 가정합니다. 기울어진 채로 SLAM을 켜면 그 기울기가 "수평"으로 저장돼 위치가 비스듬히 계산됩니다. 반드시 평평한 바닥에 놓고 SLAM을 켜세요.

검증 결과 — GPS 없이 5번 모두 완주, 위치 오차는 평균 약 0.1m

검증됨

오차가 작을수록 "드론이 자기 위치를 정확히 안다"는 뜻입니다. 0.1m = 손바닥 하나. 모두 이 문서에 실린 최종 파일로 비행했습니다.

처음부터 잘 된 것은 아닙니다. 검증 중 실패하고 고친 것들입니다. 같은 증상을 만나면 §9를 보세요.
  • 카메라 640×480 → ROS로 넘어오며 영상 대부분이 사라져 SLAM 위치가 초당 0.4개 → 320×240으로 낮춰 초당 26~30개.
  • 목표점을 2m씩 한 번에 던짐 → 급하게 움직여 SLAM이 길을 잃고 PX4가 비상 하강 → 초당 0.5m씩 옮기도록 수정.
  • EKF2 상태 플래그(1초에 한 번 발행)를 "1초 넘으면 끊김"으로 판정해 멀쩡한 비행을 중단 → 마지막 값 + 3초 기준으로 수정.
  • WSL에서 PC 전체가 1.7~2.2초씩 멈춰 GPS 비행이 "끊김"으로 중단 → 시뮬에서는 시뮬 시계로 판정.

    #!/usr/bin/env python3
"""YOLO로 카메라 영상 속 물체를 찾고, 사람이 가까이 있으면 비행을 일시정지시킨다.

입력 : image_topic (기본 /camera, 실기체 RealSense는 /camera/camera/color/image_raw)
출력 : /yolo/image (네모 박스가 그려진 영상), /yolo/detections (글자), /safety/pause (Bool)
"""
import time

import numpy as np
import rclpy
import torch
from rclpy.node import Node
from sensor_msgs.msg import Image
from std_msgs.msg import Bool, String
from ultralytics import YOLO


class YoloDetect(Node):
    def __init__(self):
        super().__init__("yolo_detect")
        self.declare_parameter("image_topic", "/camera")
        self.declare_parameter("model", "yolo11n.pt")
        self.declare_parameter("confidence", 0.5)
        self.declare_parameter("pause_class", "person")
        self.declare_parameter("pause_height_ratio", 0.4)  # 박스 높이가 화면의 40% 이상이면 '가깝다'
        self.declare_parameter("release_after_s", 1.5)
        self.conf = float(self.get_parameter("confidence").value)
        self.pause_class = self.get_parameter("pause_class").value
        self.ratio = float(self.get_parameter("pause_height_ratio").value)
        self.release_after = float(self.get_parameter("release_after_s").value)

        # GPU(CUDA)가 있으면 0번 GPU, 없으면 CPU. 조용히 바꾸지 않고 어느 쪽인지 로그로 알린다.
        self.device = 0 if torch.cuda.is_available() else "cpu"
        self.model = YOLO(self.get_parameter("model").value)
        where = torch.cuda.get_device_name(0) if self.device == 0 else "CPU (GPU를 못 찾음: 느림)"
        self.get_logger().info(f"모델 로드 완료, 계산 장치: {where}")
        self.img_pub = self.create_publisher(Image, "/yolo/image", 1)
        self.det_pub = self.create_publisher(String, "/yolo/detections", 10)
        self.pause_pub = self.create_publisher(Bool, "/safety/pause", 10)
        self.create_subscription(Image, self.get_parameter("image_topic").value, self.on_image, 1)
        self.last_close = 0.0

    def to_bgr(self, msg):
        if msg.encoding not in ("rgb8", "bgr8"):
            raise ValueError(f"지원하지 않는 영상 형식: {msg.encoding}")
        frame = np.frombuffer(msg.data, dtype=np.uint8).reshape(msg.height, msg.step)
        frame = frame[:, : msg.width * 3].reshape(msg.height, msg.width, 3)
        return frame[:, :, ::-1].copy() if msg.encoding == "rgb8" else frame.copy()

    def on_image(self, msg):
        try:
            bgr = self.to_bgr(msg)
        except ValueError as e:
            self.get_logger().error(str(e))
            return
        result = self.model(bgr, conf=self.conf, device=self.device, verbose=False)[0]
        names = result.names
        found = []
        close = False
        for box in result.boxes:
            label = names[int(box.cls)]
            x1, y1, x2, y2 = box.xyxy[0].tolist()
            found.append(f"{label}:{float(box.conf):.2f}")
            if label == self.pause_class and (y2 - y1) / msg.height >= self.ratio:
                close = True
        if close:
            self.last_close = time.monotonic()
        paused = (time.monotonic() - self.last_close) < self.release_after
        self.pause_pub.publish(Bool(data=paused))
        self.det_pub.publish(String(data=", ".join(found) if found else "(없음)"))

        drawn = result.plot()  # BGR
        out = Image()
        out.header = msg.header
        out.height, out.width = drawn.shape[:2]
        out.encoding = "bgr8"
        out.step = out.width * 3
        out.data = drawn.tobytes()
        self.img_pub.publish(out)


def main():
    rclpy.init()
    node = YoloDetect()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass


if __name__ == "__main__":
    main()

"""
import os

from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, ExecuteProcess
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node

HERE = os.path.dirname(os.path.abspath(__file__))


def generate_launch_description():
    rgbd = [
        ("rgb/image", "/camera/camera/color/image_raw"),
        ("rgb/camera_info", "/camera/camera/color/camera_info"),
        ("depth/image", "/camera/camera/aligned_depth_to_color/image_raw"),
    ]
    return LaunchDescription([
        DeclareLaunchArgument("cam_x", default_value="0.10"),
        DeclareLaunchArgument("cam_y", default_value="0.0"),
        DeclareLaunchArgument("cam_z", default_value="0.05"),

        # 1) RealSense 카메라: 컬러 + 깊이를 같은 시점으로 맞춰(align) 30Hz로
        Node(
            package="realsense2_camera", executable="realsense2_camera_node",
            namespace="camera", name="camera", output="screen",
            parameters=[{
                "enable_color": True,
                "enable_depth": True,
                "align_depth.enable": True,
                "enable_sync": True,
                "rgb_camera.color_profile": "640,480,30",
                "depth_module.depth_profile": "640,480,30",
            }],
        ),

        # 2) 기체 중심 -> 카메라 위치 (카메라 내부 좌표는 realsense 노드가 알아서 붙임)
        Node(
            package="tf2_ros", executable="static_transform_publisher",
            arguments=["--x", LaunchConfiguration("cam_x"),
                       "--y", LaunchConfiguration("cam_y"),
                       "--z", LaunchConfiguration("cam_z"),
                       "--frame-id", "base_link", "--child-frame-id", "camera_link"],
        ),

        # 3) 시각 오도메트리 -> /odom
        Node(
            package="rtabmap_odom", executable="rgbd_odometry", output="screen",
            parameters=[{
                "frame_id": "base_link",
                "odom_frame_id": "odom",
                "publish_tf": True,
                "approx_sync": False,
                "Odom/ResetCountdown": "0",
            }],
            remappings=rgbd,
        ),

        # 4) PX4 로 전달
        ExecuteProcess(
            cmd=["python3", os.path.join(HERE, "odom_to_px4.py"),
                 "--ros-args", "-p", "input_topic:=/odom"],
            output="screen",
        ),
    ])

  • Offboard 끊김 = 착륙(COM_OBL_RC_ACT 4), 조종기 끊김 동작(NAV_RCL_ACT) 확인
  • 지오펜스(QGC Safety): 실내는 반경·높이를 작게 참고블로그
  • 첫 비행은 끈으로 묶거나 그물 안에서, 고도 1m

확인된 것 검증됨

  • 새 Ubuntu 22.04(WSL2)에서 설치 스크립트 01~07 모두 성공(ROS 2 Humble, Gazebo Harmonic 8.15.0, RTAB-Map 0.23.13). 03(QGC)은 설치까지만 확인.
  • GPS 비행 2/2 완주, GPS 없는 V-SLAM 비행 5/5 완주. 위 표의 오차는 실측값.
  • EKF2가 GPS를 끄고 카메라 위치만 사용(cs_gnss_pos: false, cs_ev_*: true).
  • YOLO가 GPU에서 사람을 찾고 /safety/pause를 보냄(시험 사진 사용). 비행 중 YOLO가 죽으면 제자리 정지, 살아나면 재개.
  • 안전 로직 상태 주입 시험 16개 통과(Humble): 착륙 중 조종자 인계 시 명령 중단, 오래된 상태 불신, 시동 해제 확인, 잘못된 설정값 거부, NaN 공분산 차단.
  • 좌표 변환: 무작위 자세 1000개에서 오차 0.
